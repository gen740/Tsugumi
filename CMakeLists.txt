cmake_minimum_required(VERSION 3.28)
if(APPLE)
  # The app (Virtualization.framework, SwiftUI) requires macOS 26 or later.
  set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" TSUGUMI_VERSION LIMIT_COUNT 1)
project(Tsugumi VERSION ${TSUGUMI_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(TSUGUMI_BUILD_CORE "Build tsugumid and the Card worker" ON)
option(TSUGUMI_BUILD_TESTS "Build core RPC tests" ON)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Werror -Wformat=2 -Wimplicit-fallthrough)
endif()

find_program(TSUGUMI_FLATC flatc REQUIRED)

# hosq (header only, pinned by flake.lock) provides typed JSON schemas.
if(TSUGUMI_BUILD_CORE)
  set(HOSQ_INCLUDE_DIR "$ENV{HOSQ_INCLUDE_DIR}" CACHE PATH "hosq include directory")
  if(NOT EXISTS "${HOSQ_INCLUDE_DIR}/hosq/hosq.hh")
    message(FATAL_ERROR "hosq not found; set HOSQ_INCLUDE_DIR (nix develop does this)")
  endif()
  add_library(hosq INTERFACE)
  target_include_directories(hosq SYSTEM INTERFACE ${HOSQ_INCLUDE_DIR})
endif()

include(cmake/protocol.cmake)

find_package(flatbuffers CONFIG REQUIRED)
find_package(Threads REQUIRED)

if(NOT EMSCRIPTEN)
  include(cmake/crypto.cmake)
endif()

if(TSUGUMI_BUILD_CORE)
  include(cmake/native-dependencies.cmake)
  set(TSUGUMI_CARD_NIXPKGS_REV "$ENV{TSUGUMI_CARD_NIXPKGS_REV}" CACHE STRING "nixpkgs revision for Cards")
  set(TSUGUMI_CARD_NIXPKGS_NARHASH "$ENV{TSUGUMI_CARD_NIXPKGS_NARHASH}" CACHE STRING "nixpkgs NAR hash for Cards")
  add_subdirectory(util)
endif()
add_subdirectory(rpc)
if(TSUGUMI_BUILD_CORE)
  add_subdirectory(protocol)
endif()
add_subdirectory(client)
if(APPLE)
  foreach(input IN ITEMS DAWN_DIR WASMTIME_DEV WASMTIME_LIB WEBGPU_HEADERS)
    set(TSUGUMI_${input} "$ENV{TSUGUMI_${input}}" CACHE PATH "Dawn view host input: ${input}")
  endforeach()
  if(EXISTS "${TSUGUMI_DAWN_DIR}/lib/libwebgpu_dawn.a"
     AND EXISTS "${TSUGUMI_WASMTIME_DEV}/include/wasmtime.h"
     AND EXISTS "${TSUGUMI_WASMTIME_LIB}/lib/libwasmtime.dylib"
     AND EXISTS "${TSUGUMI_WEBGPU_HEADERS}/webgpu.yml")
    option(TSUGUMI_BUILD_VIEW_HOST_TESTS "Build the Dawn view host tests" ${TSUGUMI_BUILD_TESTS})
    if(TSUGUMI_BUILD_VIEW_HOST_TESTS)
      enable_testing()
    endif()
    add_subdirectory(ui/macos/ViewHost)
  endif()
endif()
if(TSUGUMI_BUILD_CORE)
  include(cmake/card-nix.cmake)
  add_subdirectory(tsugumid/document)
  add_subdirectory(workerd)
  add_subdirectory(tsugumid/runtime_support)
  add_subdirectory(tsugumid)
  add_subdirectory(hostd)
  add_subdirectory(cli)
  add_subdirectory(vm/guest)
  include(cmake/test-support.cmake)
  include(cmake/boundaries.cmake)
endif()

if(TSUGUMI_BUILD_TESTS AND TSUGUMI_BUILD_CORE)
  enable_testing()
  add_subdirectory(tests/core)
endif()
