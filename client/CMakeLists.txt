# This target is independent of the daemon, Nix and project storage.
add_library(tsugumi_client STATIC src/client.cc src/client_c.cc)
target_include_directories(tsugumi_client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(tsugumi_client PUBLIC tsugumi_protocol)
# Encoding and verification are header-only. In a Wasm build the host flatc
# installation supplies headers, never a native archive to the Wasm linker.
get_target_property(_flatbuffers_headers flatbuffers::flatbuffers INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(tsugumi_client SYSTEM PUBLIC ${_flatbuffers_headers})
target_compile_features(tsugumi_client PUBLIC cxx_std_20)
target_compile_options(tsugumi_client PRIVATE -Wall -Wextra -Werror -Wformat=2 -Wimplicit-fallthrough)

if(EMSCRIPTEN)
  target_sources(tsugumi_client PRIVATE src/transport_wasm.cc)
  target_compile_options(tsugumi_client PRIVATE -fexceptions)
  target_link_options(tsugumi_client PUBLIC -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -lwebsocket.js)
  add_executable(tsugumi_client_wasm src/wasm_bindings.cc)
  target_link_libraries(tsugumi_client_wasm PRIVATE tsugumi_client)
  target_link_options(tsugumi_client_wasm PRIVATE --bind --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1
    -sDYNAMIC_EXECUTION=0 -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web,worker -sNO_EXIT_RUNTIME=1)
  set_target_properties(tsugumi_client_wasm PROPERTIES OUTPUT_NAME tsugumi-client SUFFIX .js)
else()
  set(TSUGUMI_CLIENT_CURL_LIBRARY "" CACHE FILEPATH "Explicit native libcurl (for the macOS system TLS backend)")
  if(TSUGUMI_CLIENT_CURL_LIBRARY)
    add_library(CURL::libcurl UNKNOWN IMPORTED GLOBAL)
    set_target_properties(CURL::libcurl PROPERTIES IMPORTED_LOCATION "${TSUGUMI_CLIENT_CURL_LIBRARY}")
  else()
    find_package(CURL REQUIRED)
  endif()
  find_package(Threads REQUIRED)
  target_sources(tsugumi_client PRIVATE src/transport_native.cc)
  target_link_libraries(tsugumi_client PRIVATE CURL::libcurl Threads::Threads)
  add_executable(tsugumi-rpc src/main.cc)
  target_link_libraries(tsugumi-rpc PRIVATE tsugumi_client flatbuffers::flatbuffers)
  target_compile_options(tsugumi-rpc PRIVATE -Wall -Wextra -Werror -Wformat=2 -Wimplicit-fallthrough)
  install(TARGETS tsugumi_client tsugumi-rpc ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
  install(DIRECTORY include/ DESTINATION include)
endif()
